// src/components/FleetControlChart.jsx
// Requires: npm install recharts
import { useEffect, useState } from "react";
import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, Legend, ResponsiveContainer } from "recharts";
import { getMonthlyTrend } from "../api/client";

export default function FleetControlChart() {
  const [data, setData] = useState([]);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    getMonthlyTrend()
      .then(setData)
      .catch(() => setError("Monthly trend file not found. Run the Task 5 pipeline first."))
      .finally(() => setLoading(false));
  }, []);

  if (loading) return <p style={{ padding: 16 }}>Loading trend data...</p>;
  if (error) return <p style={{ padding: 16, color: "#a15c00" }}>⚠️ {error}</p>;

  const columns = Object.keys(data[0] || {});

  return (
    <div style={{ padding: 16 }}>
      <h2>📈 Fleet-wide Monthly Critical Issue Rate</h2>

      <ResponsiveContainer width="100%" height={350}>
        <LineChart data={data}>
          <CartesianGrid strokeDasharray="3 3" />
          <XAxis dataKey="YearMonth_str" />
          <YAxis />
          <Tooltip />
          <Legend />
          <Line type="monotone" dataKey="critical_rate" stroke="#1f77b4" dot={false} />
          <Line type="monotone" dataKey="roll_mean" stroke="#ff7f0e" dot={false} />
        </LineChart>
      </ResponsiveContainer>

      <div style={{ overflowX: "auto", marginTop: 24 }}>
        <table style={{ borderCollapse: "collapse", width: "100%" }}>
          <thead>
            <tr>
              {columns.map((col) => (
                <th key={col} style={{ border: "1px solid #ddd", padding: 8, background: "#fafafa" }}>{col}</th>
              ))}
            </tr>
          </thead>
          <tbody>
            {data.map((row, i) => (
              <tr key={i}>
                {columns.map((col) => (
                  <td key={col} style={{ border: "1px solid #eee", padding: 8 }}>{String(row[col])}</td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}
