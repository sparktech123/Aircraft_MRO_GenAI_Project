// src/components/Dashboard.jsx
import { useState } from "react";
import KpiRow from "./KpiRow";
import TrainingResultsTab from "./TrainingResultsTab";
import PredictionResultsTab from "./PredictionResultsTab";

const TABS = [
  "📊 EDA & Visualisations",
  "📈 Trend & Network Analysis",
  "💬 NLP & Root Cause Analysis",
  "🤖 Predictive Maintenance",
  "⚠️ Early Warning System",
];

function ComingSoon({ name }) {
  return (
    <div style={{ background: "#161b22", border: "1px dashed #21262d", borderRadius: 8, padding: "2rem", textAlign: "center", color: "#8b949e" }}>
      📊 {name} — coming in the next build phase.
    </div>
  );
}

export default function Dashboard() {
  const [view, setView] = useState("historical"); // historical | recent | comparative
  const [activeTab, setActiveTab] = useState(0);

  return (
    <div style={{ padding: 16, background: "#0d1117", minHeight: "100vh", color: "#e6edf3" }}>
      <div style={{ marginBottom: 16 }}>
        {[
          ["historical", "Historical Analysis (2021-2025)"],
          ["recent", "Recent Analysis (2026)"],
          ["comparative", "Comparative Insights"],
        ].map(([key, label]) => (
          <label key={key} style={{ marginRight: 20, cursor: "pointer" }}>
            <input type="radio" checked={view === key} onChange={() => setView(key)} /> {label}
          </label>
        ))}
      </div>

      {view === "comparative" ? (
        <ComingSoon name="Comparative Insights" />
      ) : (
        <>
          <h2>{view === "historical" ? "Historical MRO Analysis" : "Recent MRO Analysis"}</h2>
          <KpiRow dataset={view} />

          <div style={{ display: "flex", gap: 16, borderBottom: "1px solid #21262d", margin: "16px 0" }}>
            {TABS.map((tab, i) => (
              <div
                key={tab}
                onClick={() => setActiveTab(i)}
                style={{
                  paddingBottom: 8,
                  cursor: "pointer",
                  color: activeTab === i ? "#3b82f6" : "#8b949e",
                  borderBottom: activeTab === i ? "2px solid #3b82f6" : "none",
                  fontSize: 14,
                }}
              >
                {tab}
              </div>
            ))}
          </div>

          {activeTab === 0 && <ComingSoon name="EDA & Visualisations" />}
          {activeTab === 1 && <ComingSoon name="Trend & Network Analysis" />}
          {activeTab === 2 && <ComingSoon name="NLP & Root Cause Analysis" />}
          {activeTab === 3 && (view === "historical" ? <TrainingResultsTab /> : <PredictionResultsTab />)}
          {activeTab === 4 && <ComingSoon name="Early Warning System" />}
        </>
      )}
    </div>
  );
}
